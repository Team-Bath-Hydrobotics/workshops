objp = np.zeros((CHECKERBOARD[0]*CHECKERBOARD[1], 3), np.float32)
objp[:, :2] = np.mgrid[0:CHECKERBOARD[0], 0:CHECKERBOARD[1]].T.reshape(-1, 2)
objp *= SQUARE_SIZE

objpoints = []
imgpoints = []

images = sorted(glob.glob(IMAGE_PATH))
print(f"Found {len(images)} images in {IMAGE_PATH}\n")

successful = 0
failed = 0

for i, fname in enumerate(images, 1):
    print(f"[{i}/{len(images)}] {fname.split('/')[-1]:30s}", end=" ")
    img = cv2.imread(fname)
    if img is None:
        print("✗ (could not read)")
        failed += 1
        continue
    
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    
    # Use Difference of Gaussians for screen-rendered checkerboards
    blurred1 = cv2.GaussianBlur(gray, (3, 3), 0.5)
    blurred2 = cv2.GaussianBlur(gray, (15, 15), 3)
    dog = cv2.subtract(blurred1, blurred2)
    dog = cv2.normalize(dog, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)
    
    ret, corners = cv2.findChessboardCorners(
        dog,
        CHECKERBOARD,
        cv2.CALIB_CB_ADAPTIVE_THRESH + cv2.CALIB_CB_NORMALIZE_IMAGE
    )

    if ret and corners is not None:
        # Refine corner positions for sub-pixel accuracy
        criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.0001)
        corners_refined = cv2.cornerSubPix(gray, corners, (11, 11), (-1, -1), criteria)
        
        objpoints.append(objp)
        imgpoints.append(corners_refined)
        print("✓ DETECTED")
        successful += 1
    else:
        print("Not detected")
        failed += 1

print(f"\n{'='*70}")
print(f"Detection Results: {successful} successful / {failed} failed")
print(f"{'='*70}")

if len(objpoints) < 3:

def calculate_depth(baseline, focal_length, disparity):
    """
    Calculate depth using stereo triangulation.
    
    Args:
        baseline: Distance between cameras (cm)
        focal_length: Camera focal length (pixels)
        disparity: Pixel difference between images (pixels)
    
    Returns:
        depth: Distance to object (cm)
    """
    if disparity == 0:
        return float('inf')  # Avoid division by zero
    
    depth = (baseline * focal_length) / disparity
    return depth

1. Measurement Precision

Manual pixel measurements are inherently limited by human error and image resolution.

Small errors of even 1–2 pixels can result in significant depth errors, especially for objects far away.

Accuracy improves with higher-resolution images and careful point selection, but manual methods are always less precise than automated feature matching or stereo algorithms.

2. Disparity and Distance

Disparity decreases as objects get further away from the cameras because the projection difference between left and right images becomes smaller.

Smaller disparity values are harder to measure accurately, which reduces depth precision for distant objects.

Conversely, closer objects have larger disparity and can be measured more accurately.

3. Baseline Trade-off

The baseline is the distance between the stereo cameras:

Depth accuracy → A larger baseline increases sensitivity to disparity, improving depth accuracy for a given object.

Maximum measurable distance → Larger baselines allow accurate measurement of farther objects (larger disparity signal).

Minimum measurable distance → Too large a baseline can cause disparity to exceed image bounds for very close objects, making them unmeasurable.

Summary: Increasing baseline improves long-range depth precision but can create problems with very close objects.

4. Real-World Challenges in Underwater ROV Applications

Poor visibility due to turbidity, particles, or low light, which makes feature detection difficult.

Refraction through water and housings alters apparent positions, introducing calibration errors.

Limited camera resolution underwater, especially in low light.

Dynamic scenes (moving objects or currents) make manual point selection unreliable.

Lens distortion can vary over time, requiring recalibration.
